#!/bin/bash
# Warlord Finale - Automatisk arkivering (Fullt integrert)
DATE=$(date +%Y-%m-%d)

# Definer absolutte stier
TARGET_REPO="/home/wils/GitHub/panzerdroideka/Warlord-Logs-Public"
ARCHIVE_DIR="$TARGET_REPO/$DATE"
SOURCE_LOGS="/home/wils/Games/WARLORDLOTRO/Spillfiler" # Oppdatert til spillmappen din

# 0. Varm opp sudo-sesjon
sudo -v

# 1. Opprett dagens mappe
mkdir -p "$ARCHIVE_DIR/gaming"

echo "=== Warlord Finale: Starter arkivering for $DATE ==="

# 2. Generer system-telemetri
echo "Genererer system-logger..."
sudo dmesg > "$ARCHIVE_DIR/dmesg_$DATE.txt"
sudo journalctl --since "00:00:00" > "$ARCHIVE_DIR/journal_$DATE.txt"

# 3. Samle gaming-telemetri (hentet fra kamp_telemetri logg-lokasjon)
echo "Finner nyeste gaming-telemetri..."
cp /home/wils/GitHub/panzerdroideka/Wils-Linux-Legacy/Linux-Freedom-for-the/Warlord_TkG_Kernel_6.18.34-tkg--wils-v2.0/logs/kamp_historikk.log "$ARCHIVE_DIR/gaming/" 2>/dev/null

# 4. Git-operasjon
echo "Synkroniserer med skyen..."
cd "$TARGET_REPO" || exit
git add .
git commit -m "Warlord Telemetri-arkiv: $DATE - System & Gaming"
git push origin main

echo "=== Alt er arkivert! Du kan nå logge av med god samvittighet. ==="
